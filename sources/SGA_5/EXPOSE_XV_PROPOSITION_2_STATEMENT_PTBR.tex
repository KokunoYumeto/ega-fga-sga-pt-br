\begin{statement}{Proposição 2}
Sejam $S$ um pré-esquema de característica $p$ e $X$ um $S$-pré-esquema.
\begin{enumerate}[label=\alph*)]
\item O morfismo de Frobenius relativo
\[
\Fr_{X/S}:X\longrightarrow X^{(p/S)}
\]
é integral, sobrejetivo e radicial; é, portanto, um homeomorfismo universal.

\item Se $X\to S$ é uma imersão aberta, $X^{(p)}=X$ e $\Fr_{X/S}=\id_X$. Se $X\to S$ é uma imersão fechada que identifica $X$ com um subespaço anelado fechado $(|X|,\OO_S/\JJ)$ de $S$, então
\[
X^{(p)}=(|X|,\OO_S/\JJ^p)
\]
e $\Fr_{X/S}=(\id_{|X|},\varphi)$, onde $\varphi:\OO_S/\JJ^p\to\OO_S/\JJ$ é a projeção canônica.

\item Suponha que $X$ seja localmente de apresentação finita sobre $S$.
\begin{enumerate}[label=\arabic*)]
\item As três condições seguintes são equivalentes:
\[
\begin{array}{ll}
\text{(i)}& X\text{ é não ramificado sobre }S,\\
\text{(ii)}& \Fr_{X/S}\text{ é não ramificado},\\
\text{(iii)}& \Fr_{X/S}\text{ é um monomorfismo}.
\end{array}
\]
\item As três condições seguintes são igualmente equivalentes:
\[
\begin{array}{ll}
\text{(i')}& X\text{ é étale sobre }S,\\
\text{(ii')}& \Fr_{X/S}\text{ é étale},\\
\text{(iii')}& \Fr_{X/S}\text{ é um isomorfismo}.
\end{array}
\]
\end{enumerate}
\end{enumerate}
\end{statement}
