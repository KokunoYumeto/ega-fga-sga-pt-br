\begin{statement}{Proposição 8.5}
Seja $M$ um $A_X$-módulo localmente constante construtível em uma vizinhança de $Y$.
\begin{enumerate}[label=\alph*)]
\item Denotando por
\[
\gamma:H_{Y'}^*(X',M\otimes\mu^{\otimes d})\to
H^*(Y',M\otimes\mu^{\otimes(d-1)})(-2)
\]
a única aplicação definida pela relação
\[
f^*\circ f_* = j_*\circ\gamma+\operatorname{id},
\tag{8.5.1}
\]
Tem
\[
g_*\circ\gamma=0,
\tag{8.5.2}
\]
e a matriz inversa de (8.3.2) é
\[
\begin{pmatrix}
f_*&-i_*\\
-\gamma&g^*(\cdot)c_{d-1}(\check F)
\end{pmatrix}.
\tag{8.5.3}
\]
\item Para todo inteiro $p\ge0$, a sucessão
\[
0\to H^{p-2d}(Y,M(-d))
\xrightarrow{\lambda}
H^{p-2}(Y',M(-1))\oplus H^p(X,M)
\xrightarrow{\mu}
H^p(X',M)\to0,
\]
na qual $\lambda$ e $\mu$ designam as aplicações
\[
\lambda : x\longmapsto [g^*(x)c_{d-1}(\check F),-i_*(x)]
\qquad
\mu : (x,y)\mapsto j_*(x)+f^*(y)
\]
É exata. Além disso, a aplicação $\lambda$ admite como inversa pela esquerda
\[
\lambda' : (x,y)\longmapsto g_*(x).
\]
\end{enumerate}
\end{statement}
