\begin{statement}{Proposição 8.6.3}
\begin{enumerate}[label=\alph*)]
\item Para todo par $(M,N)$ de $A_X$-módulos localmente constantes construtíveis em uma vizinhança de $Y$, o diagrama
\[
\begin{tikzcd}
\nabla^*(X',M)\otimes\nabla^*(X',N)\arrow[r,"\mu\otimes\mu"]\arrow[d,"{(8.6.1)}"']&
H^*(X',M)\otimes H^*(X',N)\arrow[d,"\cup"]\\
\nabla^*(X',M\otimes N)\arrow[r,"\mu"]&
H^*(X',M\otimes N)
\end{tikzcd}
\]
é comutativo. Em particular, a aplicação sobrejetiva
\[
\mu_A=\mu:\nabla^*(X',A)\to H^*(X',A)
\]
é um morfismo de anéis, e a aplicação
\[
\mu_M=\mu:\nabla^*(X',M)\to H^*(X',M)
\]
é um morfismo de módulos sobre anéis variáveis.
\item Se se faz agir $\nabla^*(X',A)$ sobre $H^*(Y,M(-d))(-2d)$ mediante a aplicação
\[
\nabla^p(X',A)\otimes H^{q-2d}(Y,M(-d))
\to H^{p+q-2d}(Y,M(-d)),
\qquad
(y',x)\otimes y\mapsto i^*(x)\cdot y,
\]
então a aplicação $\lambda$ de 8.5 b) torna-se um morfismo de $\nabla^*(X',A)$-módulos.
\end{enumerate}
\end{statement}
