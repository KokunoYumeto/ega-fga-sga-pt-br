\medskip
\noindent\SGARefTarget{sga3:viii:corollary:1:4}\textbf{Corolário 1.4.}
\textit{Um grupo diagonalizável é reflexivo; portanto, o mesmo vale para
um grupo localmente diagonalizável. Se \(M\) e \(N\) são dois grupos
comutativos ordinários, o homomorfismo natural}
\SGARefTarget{sga3:viii:corollary:1:4:display:natural-isomorphism:01}\[
  \operatorname{Hom}_{S\text{-}\mathrm{gr}}(M_S,N_S)
  \longrightarrow
  \operatorname{Hom}_{S\text{-}\mathrm{gr}}(D(N_S),D(M_S))
\]
\textit{é bijetivo.}

\SGARefTarget{sga3:viii:corollary:1:4:discussion:01}
Daí se deduz um isomorfismo de funtores sobre \(S\):
\SGARefTarget{sga3:viii:eq:1:4:1}
\[
  \underline{\operatorname{Hom}}_{S\text{-}\mathrm{gr}}(M_S,N_S)
  \longrightarrow
  \underline{\operatorname{Hom}}_{S\text{-}\mathrm{gr}}(D(N_S),D(M_S)).
\]
Por outro lado, define-se de maneira evidente um homomorfismo
\SGARefTarget{sga3:viii:eq:1:4:2}
\[
  \underline{\operatorname{Hom}}_{\mathrm{gr}}(M,N)_S
  \longrightarrow
  \underline{\operatorname{Hom}}_{S\text{-}\mathrm{gr}}(M_S,N_S),
\]
que é sempre um monomorfismo e que é um isomorfismo desde que \(M\) seja
de tipo finito. Obtém-se assim:

\medskip
\noindent\SGARefTarget{sga3:viii:corollary:1:5}\textbf{Corolário 1.5.}
\SGARefTarget{sga3:viii:corollary:1:5:item:a}
\textit{Sejam \(M,N\) dois grupos comutativos ordinários, \(M\) de tipo
finito; então \(\underline{\operatorname{Hom}}_{S\text{-}\mathrm{gr}}
(D(N_S),D(M_S))\) é representável e tem-se um isomorfismo}
\[
  \operatorname{Hom}_{\mathrm{gr}}(M,N)_S
  \xrightarrow{\ \sim\ }
  \underline{\operatorname{Hom}}_{S\text{-}\mathrm{gr}}(D(N_S),D(M_S)).
\]

\SGARefTarget{sga3:viii:corollary:1:5:item:b}
Daí se conclui, mais geralmente, que, se \(G,H\) são localmente
diagonalizáveis, sendo \(H\) de tipo finito, então
\(\underline{\operatorname{Hom}}_{S\text{-}\mathrm{gr}}(G,H)\) é
representável.
O Corolário~\SGARefLink{sga3:viii:corollary:1:5}{1.5} fornece:

\medskip
\noindent\SGARefTarget{sga3:viii:corollary:1:6}\textbf{Corolário 1.6.}
\textit{Sob as hipóteses de
\SGARefLink{sga3:viii:corollary:1:5}{1.5}, se \(S\) é conexo, então}
\[
  \operatorname{Hom}_{S\text{-}\mathrm{gr}}(D_S(N),D_S(M))
  \xrightarrow{\ \sim\ }
  \operatorname{Hom}_{\mathrm{gr}}(M,N)
\]
\textit{e}
\[
  \operatorname{Isom}_{S\text{-}\mathrm{gr}}(D_S(N),D_S(M))
  \xrightarrow{\ \sim\ }
  \operatorname{Isom}_{\mathrm{gr}}(M,N).
\]
